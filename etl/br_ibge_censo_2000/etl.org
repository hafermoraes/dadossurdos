#+property: header-args:sh  :tangle ./etl.sh

* orquestração do processo de extração, tratamento e carga dos dados brutos do censo demográfico brasileiro de 2000

Este arquivo -- =etl.org= -- é a base de geração dos códigos nas linguagens *shell* e *GNU R* usadas no tratamento das informações dos microdados.

Arquivos com extensão *.org* correspondem ao modo Org no editor de texto GNU Emacs.

Em cada bloco de código a seguir pode haver a opção denominada *tangle*, que é responsável por realizar a exportação do conteúdo dos scripts em arquivos locais.
Blocos de código na linguagem *shell* são exportados no arquivo =./etl.sh= que é o script que orquestra a totalidade dos passos necessários para baixar, tratar e disponibilizar os microdados e o dicionário de variáveis.

O comando tangle no editor GNU Emacs pode ser executado com a combinação de teclas =C-c C-v t=, que equivale a executar o comando =org-babel-tangle=.

#+begin_src sh
create_temporary_directories(){

    echo "Criando diretórios temporários para dados brutos em $raw_data..."
    rm -rf ${raw_data}
    mkdir -p ${raw_data}

    echo "Criando diretórios temporários para dados tratados em $proc_data..."
    rm -rf ${proc_data}
    mkdir -p ${proc_data}

    echo "Criando diretórios temporários para dados colunares em $parq_data..."
    rm -rf ${parq_data}
    mkdir -p ${parq_data}
    mkdir -p ${parq_data}/dom
    mkdir -p ${parq_data}/pes
    mkdir -p ${parq_data}/fami

}

* dicionário de variáveis
** download do dicionário de variáveis e arquivos auxiliares

#+begin_src sh
fetch_variables_dictionary(){

    echo "Baixando o dicionário de variáveis de $vdict_url..."
    R --quiet --vanilla -e "curl::curl_download( '${vdict_url}', '${raw_data}/vdict.zip' )"

    echo 'Descompactando o dicionário de variáveis do arquivo =vdict.zip=...'
    unzip ${raw_data}/vdict.zip -d ${raw_data}

    echo 'Removendo da pasta temporária o arquivo compactado =vdict.zip= file...'
    rm -rf ${raw_data}/vdict.zip

    echo 'Removendo caracteres dos scripts SAS para melhor tratamento no R...'
    find ${raw_data}/SAS -type f | rename 's/ /_/g'
    # é necessário remover tabs (\t), aspas duplas, comentários (/* */) e cifrões ($)
    for f in $(find ${raw_data}/SAS -type f | grep sas); do
        cat ${f} | \
                sed 's/\t/    /g' | \
                sed 's/\$//g' | \
                sed 's/\"//g' | \
                sed 's/\/\*//g' | \
                sed 's/\*\///g' > /tmp/aux
        mv /tmp/aux $f
    done

}
#+end_src

** entendimento do dicionário

Os dicionários de variáveis do Censo de 2000 decorrem diretamente dos scripts/macros de leitura do SAS para base =dom= (domicílios), =pes= (pessoas) e =fami= (famílias).

Parte das informações como códigos de países, ocupações, atividade, religião, cursos superiores (graduação, mestrado e doutorado), códigos de mesorregiões, microrregiões e regiões metropolitanas estão disponíveis em arquivos auxiliares que precisam ser tratados individualmente antes de serem usados nas análises.

## ftp://ftp.ibge.gov.br/Censos/Censo_Demografico_2000/Microdados/1_Documentacao_20170908.zip
## era o endereço da fonte primária no momento em que este script
## foi programado em 19.10.2024.
##
## Este arquivo é então descompactado no diretório salvo na variável
## 'base_dir' a seguir
##
base_dir <- '/tmp/etl/br_ibge_censo_2000/'

## o tibble 'input' guarda o código (a_code) e o nome por extenso (a_desc)
## das respostas da variável no censo demográfico de 2000
input <-
  read_fwf(
    file = paste0(
      base_dir,
      'raw/',
      'Arquivos Auxiliares/',
      'V1004.txt'
    ),          # caminho até o arquivo
    col_positions = fwf_widths(
      c(
        4,      # 1-4   >> código da região metropolitana    =col1=
        1024    # 5-    >> nome da região metropolitana      =col2=
      ),
      c('col1', 'col2')
    ),
    ## ignora as 6 primeiras linhas do arquivo de texto
    skip = 6,
    show_col_types = FALSE,
    ## importa 100% dos dados como texto
    col_types = cols(.default = 'c'),
    ## muito provavelmente foi utilizado o sistema operacional
    ## Microsoft Windows para salvar o arquivo. Isso introduz
    ## divergências na codificação do arquivo, motivo pelo
    ## qual é necessário ler no sistema operacional GNU/Linux
    ## o arquivo com a codificação mais próxima da original
    locale = readr::locale(encoding = "ISO-8859-1")
  ) %>%
  as_tibble() %>%
  rename(
    col1 = 1,            # primeira coluna traz o código
    col2 = 2             # segunda coluna traz o nome
  ) %>%
  mutate(
    col1 = str_remove_all(string = col1, pattern = '-')
  ) %>%
  filter(
    !grepl(pattern = 'br', col1, ignore.case = TRUE)
  ) %>%
  transmute(
    q_code = 'v1004',
    a_code = col1,
    a_desc = col2
  )

expand_grid(
  base = c('domi', 'pess', 'fami'),
  input
) %>%
  write_delim(
    file = paste0(
      base_dir,
      'data/',
      'vdict_auxfiles_v1004.txt'
    ),
    delim = '|'
  )

#+begin_src R :tangle ./gnur_vardict_sasfiles.R
## gnur_vardict_sasfiles.R
##
## entendimento do dicionário de variáveis do censo demográfico brasileiro de 2000 (br_ibge_censo_2000)

library(readxl)   # read_excel
library(dplyr)    # mutate, transmute, filter, rename, bind_rows, arrange
library(stringr)  # str_split, str_to_lower, str_trim, str_locate
library(tidyr)    # unnest
library(glue)     # glue
library(readr)    # write_delim, write_lines

## caminhos até os dados do censo 2000
src_dir <- '/tmp/etl/br_ibge_censo_2000/'
raw_data <- paste0(src_dir, 'raw/')
proc_data <- paste0(src_dir, 'data/')


## variáveis auxiliares disponíveis em arquivos próprios,
## fora dos scripts/macros do SAS
aux_vars_common <- c(
  ## variáveis comuns aos microdados de domicílios,
  ## pessoas e famílias
  'v1002', 'v1003', 'v0103', 'v0104', 'v0105', 'v1004', 'areap'
  )
aux_vars_pess <- c(
  ## variáveis exclusivas dos microdados de pessoas
  'v4090', 'v4210', 'v4230', 'v4250', 'v4260', 'v4276', 'v4355',
  'v4452', 'v4462', 'v4354', 'v4219', 'v4239', 'v4269', 'v4279',
  'v4451', 'v4461', 'v1103'
)
aux_vars <- c(aux_vars_common, aux_vars_pess)



## -------------------------------------------------------------
##    base de domicílios
## -------------------------------------------------------------

## importação do dicionário de variáveis
input_domi <-
  ## o dicionário de variáveis do Censo 2000 decorre diretamente
  ## do script/macro de leitura do software SAS, não restando
  ## outra alternativa, para trabalhar com o GNU R, a não ser
  ## tratar o dicionário de variáveis como um arquivo de formato
  ## largura fixa.
  ## Originalmente o arquivo contém caracteres binários, aspas
  ## comentários (/* e */), tabulação (\t), por isso um script
  ## shell o trata antes da importação no GNU R que ocorre nas
  ## próximas linhas.
  read_fwf(
    file = paste0(raw_data, 'SAS/LE_DOMIC.sas'),
    col_positions = fwf_widths(
      c(
        6,     # 1-6   >> contador de linha               =col1=
        13,    # 7-19  >> variável e número de caracteres =col2=
        1024   # 20-   >> dicionário de variáveis         =col3=
      ),
      c('col1', 'col2', 'col3')
    ),
    ## as primeiras 20 linhas são relevantes somente para a
    ## leitura dos microdados no software SAS, motivo pelo
    ## qual essas linhas são ignoradas (puladas) na importação
    ## no GNU R.
    skip = 20,
    show_col_types = FALSE,
    ## importa 100% dos dados como texto
    col_types = cols(.default = 'c'),
    ## muito provavelmente o software SAS utilizado pelo IBGE
    ## foi utilizado no sistema operacional Microsoft Windows
    ## para salvar o script de importação. Isso introduz
    ## divergências na codificação do arquivo, motivo pelo
    ## qual é necessário ler no sistema operacional GNU/Linux
    ## o arquivo com a codificação mais próxima da original
    locale = readr::locale(encoding = "ISO-8859-1")
  )


output_domi <-
  input_domi %>%
  filter(row_number() <= n() -7) %>%
  rowwise() %>%
  mutate(
    ## q: enunciado da questão
    ## a: dicionário da resposta
    qa = ifelse( is.na(col1), 'a', 'q'),
    ## posição do espaço ' ' que separa q_code e q_desc
    col2_space = str_locate(string = col2, pattern = ' ')[1],
    ## posições inicial e final e número de caracteres nos microdados
    pos_ini = str_remove_all(string = col1, pattern = '@'),
    nchars = str_sub(string = col2, start = col2_space + 1),
    dot = str_locate(string = nchars, pattern = '[.]')[1],
    nchars = ifelse(
      is.na(dot),
      nchars,
      str_sub(string = nchars, start = 1, end = dot - 1)
    ),
    pos_fin = as.numeric(pos_ini) + as.numeric(nchars) - 1,
    ## argumento character do commando cut
    cut_arg = glue('{pos_ini}-{pos_fin}'),
    ## código e enunciado da variável
    q_code = str_sub(string = col2, start = 1, end = col2_space - 1),
    q_code = str_trim(q_code),
    q_code = str_to_lower(q_code),
    q_desc = ifelse(is.na(col1), NA, col3),
    q_desc = ifelse( !grepl('^[.]', q_desc), q_desc, str_sub( string = q_desc, start = 3)),
    q_desc = str_trim(q_desc),
    q_desc = str_to_lower(q_desc),
    ## código e descrição da resposta
    col3_hifen = str_locate(string = col3, pattern = '-')[1],
    a_code = ifelse(qa=='a', str_sub(string = col3, start = 1, end = col3_hifen - 1), NA),
    a_code = str_to_lower(a_code),
    a_desc = ifelse(qa=='a', str_sub(string = col3, start = col3_hifen + 1), NA),
    a_desc = str_trim(a_desc),
    a_desc = str_to_lower(a_desc),
    ## inclui um marcador para o caso de a variável ter seu

fetch_and_convert_microdata_to_parquet_format(){

    # Censo 2000 possui 3 arquivos por estado:
    #  - dom, base de domicílios
    #  - pes, base de pessoas
    #  - fami, base de famílias
    base="dom fami"
    #base="pes"

    for uf in PE; do # AC AL AM AP BA CE DF ES GO MA MG MS MT PA PB PE PI PR RJ RN RO RR RS SC SE SP TO; do

        echo "Baixando os microdados de ${mdata_url}/${uf}.zip..."
        R --quiet --vanilla -e "curl::curl_download( '${mdata_url}/$uf.zip', '${raw_data}/${uf}.zip' )"

        echo "Descompactando os microdados do arquivo ${uf}.zip em ${raw_data}/${uf}..."
        unzip -o ${raw_data}/$uf.zip -d ${raw_data}

        echo "Removendo da pasta temporária o arquivo compactado ${uf}.zip..."
        rm -rf ${raw_data}/$uf.zip

        echo "Tratando as bases de ${uf}..."

        for b in ${base}; do

            echo "  Base: ${b}"
            cut_args=$(cat ${proc_data}/cut_character_args_${b}.txt )

            for f in $(find ${raw_data}/ -print | grep -i txt | grep -i ${b}); do

                echo "    Convertendo para csv o arquivo ${f}..."
                cut \
                    --output-delimiter=',' \
                    --characters=$cut_args \
                    ${f} | \
                    sed -e 's/\r//g' | \
                    tr -d '\032' | \
                    tr -d ' ' > \
                    ${proc_data}/microdata_${b}_${uf}.csv

                echo "    Removendo arquivo original ${f}..."
                rm -rf ${f}

                echo "    Convertendo a base de csv para parquet em ${parq_data}/${b}/${uf}..."
                R CMD BATCH --quiet --vanilla "--args base='${b}' uf='${uf}'" ./gnur_transform_csv_into_parquet.R

                echo "    Removendo o arquivo ${proc_data}/microdata_${b}_${uf}.csv..."
                rm -rf ${proc_data}/microdata_${b}_${uf}.csv
            done
        done

    done

    # remove arquivos temporários do GNU R do diretório
    rm -rf *.{Rout,out}
}
