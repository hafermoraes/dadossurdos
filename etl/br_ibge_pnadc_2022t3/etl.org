base_url=ftp://ftp.ibge.gov.br/
base_url=${base_url}Trabalho_e_Rendimento/
base_url=${base_url}Pesquisa_Nacional_por_Amostra_de_Domicilios_continua/

base_url=${base_url}Microdados/
base_url=${base_url}Trimestre/
base_url=${base_url}Trimestre_3
mdata_url=${base_url}/Dados/PNADC_2022_trimestre3.zip
vdict_url=${base_url}/Documentacao/dicionario_PNADC_microdados_trimestre3_20240619.xls

#+begin_src sh
create_temporary_directories(){

    echo "Creating temporary directory for raw data at $raw_data..."
    rm -rf ${raw_data}
    mkdir -p ${raw_data}

    echo "DONE.\nCreating temporary directory for processed data at $proc_data..."
    rm -rf ${proc_data}
    mkdir -p ${proc_data}
    mkdir -p ${parq_data}

    echo "DONE."
}
#+end_src


#+begin_src sh
fetch_microdata(){

    echo "Fetching the =microdata= from $mdata_url..."
    R --quiet --vanilla -e "curl::curl_download( '${mdata_url}', '${raw_data}/mdata.zip' )"

    echo '  DONE. Unziping the =microdata=...'
    unzip ${raw_data}/mdata.zip -d ${raw_data}

    echo '  DONE. Deleting the downloaded =microdata= zip file...'
    rm -rf ${raw_data}/mdata.zip
    echo 'DONE.'

}
#+end_src

fetch_variables_dictionary(){

    echo "Fetching the dictionary of variables from $vdict_url..."
    R --quiet --vanilla -e "curl::curl_download( '${vdict_url}', '${raw_data}/vdict.xls' )"

    echo '  DONE.'

#+begin_src R :tangle ./gnur_parse_variables_dictionary.R
## Fetch and process microdata and variables dictionary
## of survey PNADC 2022 T3

library(readxl)   # read_excel
library(dplyr)    # mutate, filter, as_tibble, rename, fill, unique
library(readr)    # write_delim
library(tidyr)    # fill
library(stringr)  # str_pad, str_flatten, str_replace_all

src_dir <- '/tmp/etl/br_ibge_pnadc_2022t3/'
raw_data <- paste0(src_dir, 'raw/')
proc_data <- paste0(src_dir, 'data/')

dict <-
  read_excel(
    path = paste0(raw_data, 'vdict.xls'),
    sheet = 'dicionário pnad contínua ',
    range = 'A4:H2044',
    col_types = 'text',
    col_names = FALSE
  ) %>%
  rename(
    pos = 1,
    len = 2,
    var = 3,
    q_code = 4,
    q_desc = 5,
    a_code = 6,
    a_desc = 7,
    period = 8
  ) %>%
  mutate(
    part = '',
    .before = 'pos',
    q_desc = str_remove_all( q_desc, '[\r\n\"\\"]'),
    a_desc = str_remove_all( a_desc, '[\r\n\"\\"]'),
    period = str_remove_all( period, '[\r\n\"\\"]')
  ) %>%
  fill(pos, len, var, q_code, q_desc, period) %>%
  as.data.frame()

aux_part <- ''

for (i in seq_len(nrow(dict))){
  if (grepl('^parte|^variáv', dict[i,'pos'], ignore.case = TRUE)){
    aux_part <- dict[i,'pos']
  }
  dict[i,'part'] <- aux_part
}

var_dict <-
  dict %>%
  filter(
    ## ignora linhas de título, parte e módulo da pesquisa
    !grepl('[A-Za-z]+', pos, ignore.case = TRUE),
    ) %>%
  mutate(
    var = tolower(var),
    ## delimitação dos campos (início-final) como argumento da função 'cut' chamada via shell
    characters = paste0(pos, "-", as.numeric(pos) + as.numeric(len) -1)
  ) %>%
  transmute(part, q_code = var, q_desc, a_code, a_desc, characters)

## dicionário de variáveis
var_dict %>%
  select(part, q_code, q_desc, a_code, a_desc) %>%
  write_delim(
    file = paste0(src_dir, 'variables_dictionary.txt'),
    delim = '|',
    quote = 'none',
    col_names = TRUE
  )

## argumento 'characters' da função 'cut' para delimitação do microdado
var_dict %>%
  select(characters) %>%
  unique() %>%
  summarise(
    args = str_flatten(characters, ",")
  ) %>%
  write_lines(
    file = paste0(proc_data, 'cut_characters_argument.txt')
  )

## nomes das colunas dos microdados (necessário para os arquivos .parquet)
var_dict %>%
  select(q_code) %>%
  unique() %>%
  write_delim(
    file = paste0(proc_data, 'col_names.txt'),
    delim = '',
    quote = 'none',
    col_names = TRUE
  )

#+end_src


#+begin_src sh
parse_variables_dictionary_using_GNU_R(){

    echo 'Parsing the variables dictionary using GNU R'
    R CMD BATCH --quiet --vanilla ./gnur_parse_variables_dictionary.R
    rm *.Rout

    echo 'DONE.'
}
#+end_src


#+begin_src sh
prepare_microdata_for_data_ingestion(){

    echo 'Transforming microdata from fixed width format to delimited file...'

    # 'characters' argument from 'cut' function
    CHARACTERS_ARG=$(cat ${proc_data}/cut_characters_argument.txt)

    # from fixed width format to pipe (|) delimited text file
    cut \
        --output-delimiter=',' \
        --characters=$CHARACTERS_ARG \
        ${raw_data}/PNADC_2022_trimestre3.txt > \
        ${proc_data}/microdata.txt

    echo 'DONE.'
}
#+end_src


#+begin_src R :tangle ./gnur_transform_microdata_into_parquet.R
## Converte microdados do formato texto para o formato parquet usando library(arrow)
library(arrow)  # open_dataset, write_dataset
library(readr)  # read_delim
library(purrr)  # map

src_dir <- '/tmp/etl/br_ibge_pnadc_2022t3/'
raw_data <- paste0(src_dir, 'raw/')
proc_data <- paste0(src_dir, 'data/')
parq_data <- paste0(src_dir, 'parquet/')


col_names <-
  read_delim(
    file = paste0(proc_data, 'col_names.txt'),
    delim = '|',
    show_col_types = FALSE
  )
## adaptado de https://stackoverflow.com/a/71305598
csv_schema <- schema(
  purrr::map( col_names$q_code, ~Field$create(name = .x, type = string()))
)


## leitura dos microdados delimitados por ',' via
## pacote 'arrow'.
microdata_pipe_delimited <-
  open_dataset(
    paste0(proc_data, 'microdata.txt'),
    format="csv",
    schema = csv_schema,
    col_names = TRUE,
    skip = 0
  )

## escrita dos microdados tratados em formato 'parquet'
## para agilidade e menor espaço em disco
  write_parquet(
    microdata_pipe_delimited, 
    sink = paste0(parq_data, 'microdata.parquet'),
  )
#+end_src


#+begin_src sh
convert_microdata_to_parquet_using_arrow_and_GNU_R(){

    echo 'Converting the processed microdata to .parquet format using GNU R and arrow library...'
    R CMD BATCH --quiet --vanilla ./gnur_transform_microdata_into_parquet.R
    rm *.Rout

    echo 'DONE.'
}
#+end_src


#+begin_src sh
remove_temporary_steps(){

    echo "Removing the temporary intermediate files"
    rm -rf ${raw_data}
    rm -rf ${proc_data}
    echo "DONE."
}
#+end_src


#+begin_src sh
main(){

    create_temporary_directories

    fetch_microdata

    fetch_variables_dictionary

    parse_variables_dictionary_using_GNU_R

    prepare_microdata_for_data_ingestion

    convert_microdata_to_parquet_using_arrow_and_GNU_R

    remove_temporary_steps

}

main

exit 0
