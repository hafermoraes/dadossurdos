#!/bin/bash

# shell script configs
set -eu

echo "Extract-Transform Scripts for IBGE/PNS2019\n"
#+end_src


#+begin_src sh
# microdata (in text format)
mdata=ftp://ftp.ibge.gov.br/PNS/2019/Microdados/Dados/PNS_2019_20220525.zip

# dictionary of variables (needed to parse microdata)
vdict=ftp://ftp.ibge.gov.br/PNS/2019/Microdados/Documentacao/Dicionario_e_input_20220530.zip

# temporary directory
data=pns
tmpdir=/tmp/${data}/raw

# target directory
tgtdir=/tmp/$data/data
#+end_src


#+begin_src sh
check_if_dependencies_are_installed(){
    i=0

    if ! command -v curl >/dev/null 2>&1
    then
        echo "please install 'curl' to continue."
        i=1
    fi

    if ! command -v unzip >/dev/null 2>&1
    then
        echo "please install 'unzip' to continue."
        i=1
    fi

    if [ $i -gt 0 ];
    then
        echo 'please install the missing dependencies before proceeding.'
        exit 1
    fi
}
#+end_src


#+begin_src sh
create_temporary_directories(){

    echo "Creating temporary directory $tmpdir..."
    rm -rf ${tmpdir}
    mkdir -p ${tmpdir}

    echo "DONE.\nCreating temporary directory $tgtdir..."
    rm -rf ${tgtdir}
    mkdir -p ${tgtdir}

    echo "DONE."
}
#+end_src


#+begin_src sh
fetch_microdata(){

    echo "Fetching the microdata from $mdata..."
    curl ${mdata} -o ${tmpdir}/mdata.zip

    echo '  DONE. Unziping the microdata...'
    unzip ${tmpdir}/mdata.zip -d ${tmpdir}

    echo '  DONE. Deleting the downloaded microdata zip file...'
    rm -rf ${tmpdir}/mdata.zip
    echo 'DONE.'

}
#+end_src


#+begin_src sh
fetch_variables_dictionary(){
    
    echo "Fetching the dictionary of variables from $vdict..."
    curl ${vdict} -o ${tmpdir}/vdict.zip

    echo '  DONE. Unziping the data dictionary...'
    unzip ${tmpdir}/vdict.zip -d ${tmpdir}
    
    echo '  DONE. Deleting the downloaded data dictionary zip file...'
    rm -rf ${tmpdir}/vdict.zip

}
#+end_src


#+begin_src R
## bibliotecas de análise
library(readxl)   # read_excel
library(dplyr)    # mutate, filter, as_tibble, rename, fill, unique
library(readr)    # write_delim
library(stringr)  # str_pad, str_flatten, str_replace_all

dict <-
  read_excel(
    path = '/tmp/pns/raw/dicionario_PNS_microdados_2019.xls',
    sheet = 'dicionário pns',
    range = 'A5:G5224',
    col_types = 'text',
    col_names = FALSE
  ) %>%
  rename(pos = 1, len = 2, var = 3, q_code = 4, q_desc = 5, a_code = 6, a_desc = 7) %>%
  mutate(part = '', module = '', .before = "pos", q_desc = str_replace_all( q_desc, '[\r\n\"\\"]', '')) %>%
  fill(pos, len, var, q_code, q_desc) %>%
  as.data.frame()

aux_part <- ''
aux_module <- ''

for (i in seq_len(nrow(dict))){
  if (grepl('^parte|^variáv', dict[i,'pos'], ignore.case = TRUE)){
    aux_part <- dict[i,'pos']
  }
  if (grepl('^módulo', dict[i,'pos'], ignore.case = TRUE)){
    aux_module <- dict[i,'pos']
  }
  dict[i,'part'] <- aux_part
  dict[i,'module'] <- aux_module
  ## TODO tratar final da tabela (sai antroprometria, entram variaveis derivadas/amostragem)
}


var_dict <-
  dict %>%
  filter(
    ## ignora linhas de título, parte e módulo da pesquisa
    !grepl('^parte|^variáve|^módulo|caso de mais|variáve', pos, ignore.case = TRUE),
  ) %>%
  mutate(
    var = tolower(var),
    ## delimitação dos campos (início-final) como argumento da função 'cut' chamada via shell
    characters = paste0(pos, "-", as.numeric(pos) + as.numeric(len) -1),
    ## corpo da chamada de criação da tabela (create table) contendo campo, tipo de dados e comentários
    create_table_fields = paste0(
      ## 4 espaços à esquerda
      ## completa com espaço à direita para garantir comprimento uniforme de 10 caracteres
      str_pad(string = tolower(var), width = 15, side = 'left', pad = ' '),
      ## comentários de linha no script sql (tamanho do campo e enunciado da pergunta na pesquisa)
      ifelse(var == 'vddata', " text  -- ", " text, -- "),
      q_desc
    )
  ) %>% 
  transmute(part, module, var, q_code, q_desc, a_code, a_desc, characters, create_table_fields) 


## argumento 'characters' da função 'cut' para delimitação do microdado
var_dict %>%
  select(characters) %>%
  unique() %>%
  summarise(
    args = str_flatten(characters, ",")
  ) %>%
  write_lines(
    file = '/tmp/pns/data/cut_characters'
  )

## corpo da 'create table' contendo os nomes de coluna, tipo de dados e comentários com enunciado da pergunta da pns 2019
var_dict %>%
  as_tibble() %>%
  select(create_table_fields) %>%
  unique() %>%
  write_delim(
    file = '/tmp/pns/data/create_table_fields',
    delim='',
    quote = 'none',
    col_names = FALSE
  )


#+end_src


#+begin_src sh
main(){

    check_if_dependencies_are_installed

    create_temporary_directories
    
    fetch_microdata
    
    fetch_variables_dictionary
}

main

exit 0
#+end_src
